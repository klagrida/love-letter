enum class CardType(val value: Int, val count: Int) {
    GUARD(1, 5),
    PRIEST(2, 2),
    BARON(3, 2),
    HANDMAID(4, 2),
    PRINCE(5, 2),
    KING(6, 1),
    COUNTESS(7, 1),
    PRINCESS(8, 1);
    
    fun getDescription(): String = when(this) {
        GUARD -> "Guess a player's card (not Guard)"
        PRIEST -> "Look at another player's card"
        BARON -> "Compare cards; lower is eliminated"
        HANDMAID -> "Protection until your next turn"
        PRINCE -> "Target discards and draws new card"
        KING -> "Trade cards with another player"
        COUNTESS -> "Must discard with King/Prince"
        PRINCESS -> "Lose if you discard this"
    }
}

data class Card(val type: CardType, val id: String = "${type.name}_${System.currentTimeMillis()}")

data class Player(
    val id: String,
    val name: String,
    var hand: Card? = null,
    var isProtected: Boolean = false,
    var isEliminated: Boolean = false,
    val discardedCards: MutableList<Card> = mutableListOf(),
    var tokensOfAffection: Int = 0
)

sealed class GameAction {
    data class PlayGuard(val targetPlayerId: String, val guessedCard: CardType) : GameAction()
    data class PlayPriest(val targetPlayerId: String) : GameAction()
    data class PlayBaron(val targetPlayerId: String) : GameAction()
    object PlayHandmaid : GameAction()
    data class PlayPrince(val targetPlayerId: String) : GameAction()
    data class PlayKing(val targetPlayerId: String) : GameAction()
    object PlayCountess : GameAction()
}

data class GameState(
    val players: List<Player>,
    val deck: MutableList<Card>,
    val currentPlayerIndex: Int = 0,
    val drawnCard: Card? = null,
    val roundWinner: Player? = null,
    val gameWinner: Player? = null,
    val burnedCard: Card? = null,
    val revealedCards: List<Card> = emptyList() // For 2-player variant
)

class LoveLetterGame(playerNames: List<String>) {
    private var state: GameState
    private val requiredTokens = when(playerNames.size) {
        2 -> 7
        3 -> 5
        4 -> 4
        else -> 4
    }
    
    init {
        val players = playerNames.mapIndexed { idx, name -> 
            Player(id = "player_$idx", name = name)
        }
        state = GameState(
            players = players,
            deck = mutableListOf(),
            currentPlayerIndex = 0
        )
        startNewRound()
    }
    
    private fun createDeck(): MutableList<Card> {
        val deck = mutableListOf<Card>()
        CardType.entries.forEach { cardType ->
            repeat(cardType.count) {
                deck.add(Card(cardType))
            }
        }
        return deck.shuffled().toMutableList()
    }
    
    private fun startNewRound() {
        // Reset players
        state.players.forEach { player ->
            player.hand = null
            player.isProtected = false
            player.isEliminated = false
            player.discardedCards.clear()
        }
        
        // Create and shuffle deck
        val deck = createDeck()
        
        // Burn one card face down
        val burnedCard = deck.removeAt(0)
        
        // For 2 players, reveal 3 additional cards
        val revealedCards = if (state.players.size == 2) {
            List(3) { deck.removeAt(0) }
        } else emptyList()
        
        // Deal one card to each player
        state.players.forEach { player ->
            player.hand = deck.removeAt(0)
        }
        
        state = state.copy(
            deck = deck,
            drawnCard = null,
            roundWinner = null,
            burnedCard = burnedCard,
            revealedCards = revealedCards
        )
    }
    
    fun getState(): GameState = state
    
    fun getCurrentPlayer(): Player = state.players[state.currentPlayerIndex]
    
    fun drawCard(): Boolean {
        if (state.deck.isEmpty()) {
            endRound()
            return false
        }
        
        // Clear protection from previous turn
        getCurrentPlayer().isProtected = false
        
        val drawnCard = state.deck.removeAt(0)
        state = state.copy(drawnCard = drawnCard)
        return true
    }
    
    fun canPlayCard(card: Card): Boolean {
        val player = getCurrentPlayer()
        val hand = player.hand ?: return false
        val drawn = state.drawnCard ?: return false
        
        // Check if Countess must be discarded
        if (hand.type == CardType.COUNTESS || drawn.type == CardType.COUNTESS) {
            val otherCard = if (hand.type == CardType.COUNTESS) drawn else hand
            if (otherCard.type == CardType.KING || otherCard.type == CardType.PRINCE) {
                return card.type == CardType.COUNTESS
            }
        }
        
        return card == hand || card == drawn
    }
    
    fun playCard(card: Card, action: GameAction?): String {
        val player = getCurrentPlayer()
        val hand = player.hand ?: return "No card in hand"
        val drawn = state.drawnCard ?: return "No drawn card"
        
        if (!canPlayCard(card)) {
            return "Cannot play this card"
        }
        
        // Keep the other card in hand
        player.hand = if (card == hand) drawn else hand
        player.discardedCards.add(card)
        
        // Execute card effect
        val result = executeCardEffect(card, action)
        
        // Check if player discarded Princess
        if (card.type == CardType.PRINCESS) {
            player.isEliminated = true
        }
        
        // Move to next player or end round
        moveToNextPlayer()
        
        return result
    }
    
    private fun executeCardEffect(card: Card, action: GameAction?): String {
        return when (card.type) {
            CardType.GUARD -> {
                if (action is GameAction.PlayGuard) {
                    val target = state.players.find { it.id == action.targetPlayerId }
                    if (target == null || target.isEliminated || target.isProtected) {
                        "Invalid target"
                    } else {
                        if (target.hand?.type == action.guessedCard) {
                            target.isEliminated = true
                            "${target.name} is eliminated! They had ${action.guessedCard.name}"
                        } else {
                            "${target.name} does not have ${action.guessedCard.name}"
                        }
                    }
                } else "Guard requires a target and guess"
            }
            
            CardType.PRIEST -> {
                if (action is GameAction.PlayPriest) {
                    val target = state.players.find { it.id == action.targetPlayerId }
                    if (target == null || target.isEliminated || target.isProtected) {
                        "Invalid target"
                    } else {
                        "You see ${target.name} has ${target.hand?.type?.name}"
                    }
                } else "Priest requires a target"
            }
            
            CardType.BARON -> {
                if (action is GameAction.PlayBaron) {
                    val target = state.players.find { it.id == action.targetPlayerId }
                    val player = getCurrentPlayer()
                    if (target == null || target.isEliminated || target.isProtected) {
                        "Invalid target"
                    } else {
                        val playerValue = player.hand?.type?.value ?: 0
                        val targetValue = target.hand?.type?.value ?: 0
                        when {
                            playerValue > targetValue -> {
                                target.isEliminated = true
                                "${target.name} is eliminated!"
                            }
                            targetValue > playerValue -> {
                                player.isEliminated = true
                                "You are eliminated!"
                            }
                            else -> "Tie! No one is eliminated"
                        }
                    }
                } else "Baron requires a target"
            }
            
            CardType.HANDMAID -> {
                getCurrentPlayer().isProtected = true
                "You are protected until your next turn"
            }
            
            CardType.PRINCE -> {
                if (action is GameAction.PlayPrince) {
                    val target = state.players.find { it.id == action.targetPlayerId }
                    if (target == null || target.isEliminated) {
                        "Invalid target"
                    } else if (target.isProtected && target != getCurrentPlayer()) {
                        "Target is protected"
                    } else {
                        val discarded = target.hand
                        target.discardedCards.add(discarded!!)
                        
                        if (discarded.type == CardType.PRINCESS) {
                            target.isEliminated = true
                            "${target.name} discarded the Princess and is eliminated!"
                        } else {
                            target.hand = if (state.deck.isNotEmpty()) {
                                state.deck.removeAt(0)
                            } else {
                                state.burnedCard
                            }
                            "${target.name} discarded ${discarded.type.name} and drew a new card"
                        }
                    }
                } else "Prince requires a target"
            }
            
            CardType.KING -> {
                if (action is GameAction.PlayKing) {
                    val target = state.players.find { it.id == action.targetPlayerId }
                    val player = getCurrentPlayer()
                    if (target == null || target.isEliminated || target.isProtected) {
                        "Invalid target"
                    } else {
                        val temp = player.hand
                        player.hand = target.hand
                        target.hand = temp
                        "Traded cards with ${target.name}"
                    }
                } else "King requires a target"
            }
            
            CardType.COUNTESS -> "Countess discarded"
            CardType.PRINCESS -> "You are eliminated!"
        }
    }
    
    private fun moveToNextPlayer() {
        state = state.copy(drawnCard = null)
        
        // Check if round should end
        val activePlayers = state.players.filter { !it.isEliminated }
        
        if (activePlayers.size <= 1) {
            endRound()
            return
        }
        
        if (state.deck.isEmpty()) {
            endRound()
            return
        }
        
        // Find next active player
        var nextIndex = (state.currentPlayerIndex + 1) % state.players.size
        while (state.players[nextIndex].isEliminated) {
            nextIndex = (nextIndex + 1) % state.players.size
        }
        
        state = state.copy(currentPlayerIndex = nextIndex)
    }
    
    private fun endRound() {
        val activePlayers = state.players.filter { !it.isEliminated }
        
        val winner = if (activePlayers.size == 1) {
            activePlayers.first()
        } else {
            // Highest card wins, ties broken by discard pile value
            activePlayers.maxByOrNull { player ->
                val cardValue = player.hand?.type?.value ?: 0
                val discardValue = player.discardedCards.sumOf { it.type.value }
                cardValue * 1000 + discardValue
            }
        }
        
        winner?.tokensOfAffection = (winner?.tokensOfAffection ?: 0) + 1
        
        state = state.copy(roundWinner = winner)
        
        // Check for game winner
        if (winner?.tokensOfAffection == requiredTokens) {
            state = state.copy(gameWinner = winner)
        }
    }
    
    fun startNextRound() {
        if (state.gameWinner == null) {
            startNewRound()
        }
    }
    
    fun getAvailableTargets(excludeSelf: Boolean = true): List<Player> {
        val currentPlayer = getCurrentPlayer()
        return state.players.filter { player ->
            !player.isEliminated && 
            !player.isProtected && 
            (!excludeSelf || player.id != currentPlayer.id)
        }
    }
}

import androidx.compose.foundation.background

@Composable
fun LoveLetterApp() {
    var gameStarted by remember { mutableStateOf(false) }
    var game by remember { mutableStateOf<LoveLetterGame?>(null) }
    
    if (!gameStarted) {
        StartScreen(onStartGame = { playerNames ->
            game = LoveLetterGame(playerNames)
            gameStarted = true
        })
    } else {
        game?.let { GameScreen(it) }
    }
}

@Composable
fun StartScreen(onStartGame: (List<String>) -> Unit) {
    var numPlayers by remember { mutableStateOf(2) }
    val playerNames = remember { mutableStateListOf("Player 1", "Player 2") }
    
    MaterialTheme {
        Column(
            modifier = Modifier
                .fillMaxSize()
                .background(Color(0xFFFFE4E1))
                .padding(24.dp)
                .verticalScroll(rememberScrollState()),
            horizontalAlignment = Alignment.CenterHorizontally,
            verticalArrangement = Arrangement.Center
        ) {
            Text(
                "💌 Love Letter",
                style = MaterialTheme.typography.displayMedium,
                fontWeight = FontWeight.Bold,
                color = Color(0xFFD32F2F)
            )
            
            Spacer(modifier = Modifier.height(32.dp))
            
            Text("Number of Players:", fontWeight = FontWeight.Bold)
            Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                (2..4).forEach { num ->
                    Button(
                        onClick = {
                            numPlayers = num
                            while (playerNames.size < num) {
                                playerNames.add("Player ${playerNames.size + 1}")
                            }
                            while (playerNames.size > num) {
                                playerNames.removeAt(playerNames.size - 1)
                            }
                        },
                        colors = if (num == numPlayers) {
                            ButtonDefaults.buttonColors()
                        } else {
                            ButtonDefaults.outlinedButtonColors()
                        }
                    ) {
                        Text("$num")
                    }
                }
            }
            
            Spacer(modifier = Modifier.height(24.dp))
            
            playerNames.forEachIndexed { idx, name ->
                OutlinedTextField(
                    value = name,
                    onValueChange = { playerNames[idx] = it },
                    label = { Text("Player ${idx + 1}") },
                    modifier = Modifier.fillMaxWidth()
                )
                Spacer(modifier = Modifier.height(8.dp))
            }
            
            Spacer(modifier = Modifier.height(24.dp))
            
            Button(
                onClick = { onStartGame(playerNames.toList()) },
                modifier = Modifier.fillMaxWidth()
            ) {
                Text("Start Game", style = MaterialTheme.typography.titleMedium)
            }
        }
    }
}

@Composable
fun GameScreen(game: LoveLetterGame) {
    val state = game.getState()
    val currentPlayer = game.getCurrentPlayer()
    var selectedCard by remember { mutableStateOf<Card?>(null) }
    var selectedAction by remember { mutableStateOf<GameAction?>(null) }
    var message by remember { mutableStateOf("") }
    var showCardSelect by remember { mutableStateOf(false) }
    
    MaterialTheme {
        Column(
            modifier = Modifier
                .fillMaxSize()
                .background(Color(0xFFFFF0F5))
                .padding(16.dp)
                .verticalScroll(rememberScrollState())
        ) {
            // Header
            Text(
                "💌 Love Letter",
                style = MaterialTheme.typography.headlineMedium,
                fontWeight = FontWeight.Bold,
                color = Color(0xFFD32F2F)
            )
            
            // Player Scores
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceEvenly
            ) {
                state.players.forEach { player ->
                    Column(horizontalAlignment = Alignment.CenterHorizontally) {
                        Text(
                            player.name,
                            fontWeight = if (player == currentPlayer) FontWeight.Bold else FontWeight.Normal,
                            color = if (player.isEliminated) Color.Gray else Color.Black
                        )
                        Text("❤️ ${player.tokensOfAffection}")
                    }
                }
            }
            
            Divider(modifier = Modifier.padding(vertical = 16.dp))
            
            // Round Winner or Game Winner
            state.gameWinner?.let { winner ->
                Card(
                    modifier = Modifier.fillMaxWidth(),
                    colors = CardDefaults.cardColors(containerColor = Color(0xFFFFD700))
                ) {
                    Column(
                        modifier = Modifier.padding(16.dp),
                        horizontalAlignment = Alignment.CenterHorizontally
                    ) {
                        Text(
                            "🏆 ${winner.name} Wins!",
                            style = MaterialTheme.typography.headlineSmall,
                            fontWeight = FontWeight.Bold
                        )
                        Spacer(modifier = Modifier.height(8.dp))
                        Button(onClick = { /* Restart */ }) {
                            Text("New Game")
                        }
                    }
                }
                return@Column
            }
            
            state.roundWinner?.let { winner ->
                Card(
                    modifier = Modifier.fillMaxWidth(),
                    colors = CardDefaults.cardColors(containerColor = Color(0xFF90EE90))
                ) {
                    Column(modifier = Modifier.padding(16.dp)) {
                        Text(
                            "Round Winner: ${winner.name}",
                            fontWeight = FontWeight.Bold,
                            style = MaterialTheme.typography.titleLarge
                        )
                        Button(onClick = { 
                            game.startNextRound()
                            selectedCard = null
                            message = ""
                        }) {
                            Text("Next Round")
                        }
                    }
                }
                return@Column
            }
            
            // Current Player Turn
            Card(
                modifier = Modifier.fillMaxWidth(),
                colors = CardDefaults.cardColors(containerColor = Color.White)
            ) {
                Column(modifier = Modifier.padding(16.dp)) {
                    Text(
                        "Current Turn: ${currentPlayer.name}",
                        style = MaterialTheme.typography.titleLarge,
                        fontWeight = FontWeight.Bold
                    )
                    
                    if (currentPlayer.isProtected) {
                        Text("🛡️ Protected", color = Color(0xFF2196F3))
                    }
                }
            }
            
            Spacer(modifier = Modifier.height(16.dp))
            
            // Draw Card Button
            if (state.drawnCard == null && !currentPlayer.isEliminated) {
                Button(
                    onClick = {
                        val success = game.drawCard()
                        if (success) {
                            message = "Card drawn!"
                        } else {
                            message = "Round ended"
                        }
                    },
                    modifier = Modifier.fillMaxWidth()
                ) {
                    Text("Draw Card")
                }
            }
            
            // Show Cards
            if (state.drawnCard != null) {
                Text("Your Cards:", fontWeight = FontWeight.Bold)
                Spacer(modifier = Modifier.height(8.dp))
                
                Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                    currentPlayer.hand?.let { card ->
                        CardView(
                            card = card,
                            isSelected = selectedCard == card,
                            onClick = { selectedCard = card; showCardSelect = true }
                        )
                    }
                    
                    state.drawnCard?.let { card ->
                        CardView(
                            card = card,
                            isSelected = selectedCard == card,
                            onClick = { selectedCard = card; showCardSelect = true }
                        )
                    }
                }
            }
            
            // Action Selection Dialog
            if (showCardSelect && selectedCard != null) {
                Spacer(modifier = Modifier.height(16.dp))
                Card(
                    modifier = Modifier.fillMaxWidth(),
                    colors = CardDefaults.cardColors(containerColor = Color(0xFFE3F2FD))
                ) {
                    Column(modifier = Modifier.padding(16.dp)) {
                        Text(
                            "Playing: ${selectedCard!!.type.name}",
                            fontWeight = FontWeight.Bold
                        )
                        Text(selectedCard!!.type.getDescription())
                        
                        Spacer(modifier = Modifier.height(8.dp))
                        
                        // Action selection based on card type
                        when (selectedCard!!.type) {
                            CardType.GUARD -> GuardActionSelector(game, onActionSelected = { selectedAction = it })
                            CardType.PRIEST, CardType.BARON, CardType.KING -> 
                                TargetSelector(game, onActionSelected = { selectedAction = it })
                            CardType.PRINCE -> PrinceSelector(game, onActionSelected = { selectedAction = it })
                            CardType.HANDMAID, CardType.COUNTESS, CardType.PRINCESS -> 
                                selectedAction = when(selectedCard!!.type) {
                                    CardType.HANDMAID -> GameAction.PlayHandmaid
                                    CardType.COUNTESS -> GameAction.PlayCountess
                                    else -> null
                                }
                        }
                        
                        Spacer(modifier = Modifier.height(8.dp))
                        
                        Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                            Button(
                                onClick = {
                                    val result = game.playCard(selectedCard!!, selectedAction)
                                    message = result
                                    selectedCard = null
                                    selectedAction = null
                                    showCardSelect = false
                                },
                                enabled = selectedAction != null || selectedCard!!.type in listOf(
                                    CardType.HANDMAID, CardType.COUNTESS, CardType.PRINCESS
                                )
                            ) {
                                Text("Play Card")
                            }
                            
                            OutlinedButton(onClick = {
                                selectedCard = null
                                selectedAction = null
                                showCardSelect = false
                            }) {
                                Text("Cancel")
                            }
                        }
                    }
                }
            }
            
            // Message Display
            if (message.isNotEmpty()) {
                Spacer(modifier = Modifier.height(16.dp))
                Card(
                    colors = CardDefaults.cardColors(containerColor = Color(0xFFFFF9C4))
                ) {
                    Text(message, modifier = Modifier.padding(12.dp))
                }
            }
            
            // Game Info
            Spacer(modifier = Modifier.height(16.dp))
            Text("Cards remaining in deck: ${state.deck.size}", color = Color.Gray)
        }
    }
}

@Composable
fun CardView(card: Card, isSelected: Boolean, onClick: () -> Unit) {
    Card(
        modifier = Modifier
            .width(100.dp)
            .height(140.dp)
            .clickable(onClick = onClick)
            .border(
                width = if (isSelected) 3.dp else 1.dp,
                color = if (isSelected) Color(0xFFD32F2F) else Color.Gray,
                shape = RoundedCornerShape(8.dp)
            ),
        shape = RoundedCornerShape(8.dp)
    ) {
        Column(
            modifier = Modifier
                .fillMaxSize()
                .background(Color.White)
                .padding(8.dp),
            horizontalAlignment = Alignment.CenterHorizontally,
            verticalArrangement = Arrangement.Center
        ) {
            Text(
                "${card.type.value}",
                style = MaterialTheme.typography.headlineMedium,
                fontWeight = FontWeight.Bold,
                color = Color(0xFFD32F2F)
            )
            Text(
                card.type.name,
                style = MaterialTheme.typography.bodySmall,
                textAlign = TextAlign.Center
            )
        }
    }
}

@Composable
fun GuardActionSelector(game: LoveLetterGame, onActionSelected: (GameAction) -> Unit) {
    var selectedTarget by remember { mutableStateOf<Player?>(null) }
    var selectedGuess by remember { mutableStateOf<CardType?>(null) }
    val targets = game.getAvailableTargets()
    
    Column {
        Text("Select target:")
        targets.forEach { player ->
            Row(
                modifier = Modifier
                    .fillMaxWidth()
                    .clickable { selectedTarget = player }
                    .padding(8.dp),
                verticalAlignment = Alignment.CenterVertically
            ) {
                RadioButton(
                    selected = selectedTarget == player,
                    onClick = { selectedTarget = player }
                )
                Text(player.name)
            }
        }
        
        if (selectedTarget != null) {
            Text("Guess card:")
            CardType.entries.filter { it != CardType.GUARD }.forEach { cardType ->
                Row(
                    modifier = Modifier
                        .fillMaxWidth()
                        .clickable { 
                            selectedGuess = cardType
                            onActionSelected(GameAction.PlayGuard(selectedTarget!!.id, cardType))
                        }
                        .padding(8.dp),
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    RadioButton(
                        selected = selectedGuess == cardType,
                        onClick = {
                            selectedGuess = cardType
                            onActionSelected(GameAction.PlayGuard(selectedTarget!!.id, cardType))
                        }
                    )
                    Text("${cardType.value} - ${cardType.name}")
                }
            }
        }
    }
}

@Composable
fun TargetSelector(game: LoveLetterGame, onActionSelected: (GameAction) -> Unit) {
    val targets = game.getAvailableTargets()
    val currentCard = game.getState().drawnCard?.type ?: game.getCurrentPlayer().hand?.type
    
    Text("Select target:")
    targets.forEach { player ->
        Button(
            onClick = {
                val action = when(currentCard) {
                    CardType.PRIEST -> GameAction.PlayPriest(player.id)
                    CardType.BARON -> GameAction.PlayBaron(player.id)
                    CardType.KING -> GameAction.PlayKing(player.id)
                    else -> return@Button
                }
                onActionSelected(action)
            },
            modifier = Modifier.fillMaxWidth()
        ) {
            Text(player.name)
        }
    }
}

@Composable
fun PrinceSelector(game: LoveLetterGame, onActionSelected: (GameAction) -> Unit) {
    val targets = game.getAvailableTargets(excludeSelf = false)
    
    Text("Select target (can be yourself):")
    targets.forEach { player ->
        Button(
            onClick = { onActionSelected(GameAction.PlayPrince(player.id)) },
            modifier = Modifier.fillMaxWidth()
        ) {
            Text(player.name)
        }
    }
}
